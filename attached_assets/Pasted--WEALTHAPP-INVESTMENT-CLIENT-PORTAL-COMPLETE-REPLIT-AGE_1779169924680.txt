### src/lib/investmentGoalProjection.ts

```typescript
export interface PackageInput {
  currentValueUsd: number
  monthlyContribution: number  // 0 if lump sum only
  expectedAnnualReturn: number // advisor-set on the package
  type: 'rsp' | 'lump_sum' | 'combination'
}

export interface InvestmentGoalProjectionResult {
  status: 'on_track' | 'almost' | 'off_track'
  onTrack: boolean
  monthsRemaining: number
  totalCurrentValue: number
  totalProjectedValue: number
  totalMonthlyContribution: number
  projectedShortfall: number
  gapPercent: number
  amountStillNeeded: number
  monthlyShortfall: number
  additionalMonthlyNeeded: number  // extra RSP to get on track
  lumpSumNeededToday: number       // lump sum to get on track
}

export function calculateInvestmentGoalProjection(
  targetAmount: number,
  targetDate: string,
  packages: PackageInput[]
): InvestmentGoalProjectionResult {
  const today = new Date()
  const target = new Date(targetDate)
  const monthsRemaining = Math.max(0,
    (target.getFullYear() - today.getFullYear()) * 12 +
    (target.getMonth() - today.getMonth())
  )

  let totalCurrentValue = 0
  let totalProjectedValue = 0
  let totalMonthlyContribution = 0

  for (const pkg of packages) {
    totalCurrentValue += pkg.currentValueUsd
    totalMonthlyContribution += pkg.monthlyContribution
    const monthlyRate = pkg.expectedAnnualReturn / 100 / 12

    // Future value of current balance
    const fvBalance = pkg.currentValueUsd * Math.pow(1 + monthlyRate, monthsRemaining)
    // Future value of regular monthly contributions (annuity formula)
    const fvContributions = (pkg.monthlyContribution > 0 && monthlyRate > 0)
      ? pkg.monthlyContribution * ((Math.pow(1 + monthlyRate, monthsRemaining) - 1) / monthlyRate)
      : pkg.monthlyContribution * monthsRemaining

    totalProjectedValue += fvBalance + fvContributions
  }

  const projectedShortfall = Math.max(0, targetAmount - totalProjectedValue)
  const gapPercent = targetAmount > 0 ? (projectedShortfall / targetAmount) * 100 : 0
  const amountStillNeeded = Math.max(0, targetAmount - totalCurrentValue)
  const requiredMonthly = monthsRemaining > 0 ? amountStillNeeded / monthsRemaining : 0
  const monthlyShortfall = Math.max(0, requiredMonthly - totalMonthlyContribution)

  // Average return rate across packages for gap calculations
  const avgAnnualReturn = packages.length > 0
    ? packages.reduce((sum, p) => sum + p.expectedAnnualReturn, 0) / packages.length
    : 7.0
  const avgMonthlyRate = avgAnnualReturn / 100 / 12

  // How much extra monthly RSP closes the gap?
  const additionalMonthlyNeeded = projectedShortfall > 0 && monthsRemaining > 0 && avgMonthlyRate > 0
    ? projectedShortfall * avgMonthlyRate / (Math.pow(1 + avgMonthlyRate, monthsRemaining) - 1)
    : projectedShortfall / Math.max(monthsRemaining, 1)

  // What lump sum today closes the gap?
  const lumpSumNeededToday = projectedShortfall > 0 && monthsRemaining > 0
    ? projectedShortfall / Math.pow(1 + avgMonthlyRate, monthsRemaining)
    : projectedShortfall

  const onTrack = totalProjectedValue >= targetAmount
  const status = onTrack ? 'on_track' : gapPercent <= 15 ? 'almost' : 'off_track'

  return {
    status, onTrack, monthsRemaining,
    totalCurrentValue, totalProjectedValue, totalMonthlyContribution,
    projectedShortfall, gapPercent, amountStillNeeded,
    monthlyShortfall, additionalMonthlyNeeded, lumpSumNeededToday
  }
}

// Scenario calculations

export function scenarioIncreaseMonthly(
  currentPackages: PackageInput[],
  additionalMonthly: number,
  targetAmount: number,
  targetDate: string,
  monthsRemaining: number
) {
  const enhanced = currentPackages.map((pkg, i) =>
    i === 0 // apply increase to first RSP package (or all if preference)
      ? { ...pkg, monthlyContribution: pkg.monthlyContribution + additionalMonthly }
      : pkg
  )
  return calculateInvestmentGoalProjection(targetAmount, targetDate, enhanced)
}

export function scenarioAddLumpSum(
  currentPackages: PackageInput[],
  lumpSumAmount: number,
  targetAmount: number,
  targetDate: string
) {
  const enhanced = currentPackages.map((pkg, i) =>
    i === 0
      ? { ...pkg, currentValueUsd: pkg.currentValueUsd + lumpSumAmount }
      : pkg
  )
  return calculateInvestmentGoalProjection(targetAmount, targetDate, enhanced)
}

export function scenarioMarketDrop(
  currentPackages: PackageInput[],
  dropPercent: number,  // e.g. 20 for a 20% drop
  targetAmount: number,
  targetDate: string
) {
  const factor = 1 - (dropPercent / 100)
  const dropped = currentPackages.map(pkg => ({
    ...pkg,
    currentValueUsd: pkg.currentValueUsd * factor
  }))
  const result = calculateInvestmentGoalProjection(targetAmount, targetDate, dropped)

  // Recovery estimate: months to recover at expected return
  const totalCurrentLost = currentPackages.reduce((sum, p) => sum + p.currentValueUsd, 0)
    * (dropPercent / 100)
  const avgMonthlyRate = (currentPackages.reduce((sum, p) => sum + p.expectedAnnualReturn, 0)
    / currentPackages.length) / 100 / 12
  const recoveryMonths = avgMonthlyRate > 0
    ? Math.log(1 + (dropPercent / 100)) / Math.log(1 + avgMonthlyRate)
    : 0

  return { ...result, recoveryMonths: Math.ceil(recoveryMonths), dollarLost: totalCurrentLost }
}

export function scenarioPauseContributions(
  currentPackages: PackageInput[],
  pauseMonths: number,
  targetAmount: number,
  targetDate: string
) {
  // Calculate cost of pausing = what compounding you miss
  const paused = currentPackages.map(pkg => ({
    ...pkg,
    monthlyContribution: 0  // zero during pause
  }))
  const pausedResult = calculateInvestmentGoalProjection(targetAmount, targetDate, paused)
  const normalResult = calculateInvestmentGoalProjection(targetAmount, targetDate, currentPackages)

  const pauseCost = normalResult.totalProjectedValue - pausedResult.totalProjectedValue
  const goalDelayMonths = pausedResult.onTrack ? 0
    : Math.ceil(pausedResult.projectedShortfall / (normalResult.totalMonthlyContribution || 1))

  return { ...pausedResult, pauseCost, goalDelayMonths }
}
```

```typescript
// Fetches fund prices from Twelve Data API
// Called by: Supabase Edge Function (daily cron) AND admin manual trigger

const TWELVE_DATA_BASE = 'https://api.twelvedata.com'

export async function fetchFundPrices(apiKey: string, symbols: string[]) {
  const symbolStr = symbols.join(',')
  const response = await fetch(
    `${TWELVE_DATA_BASE}/quote?symbol=${symbolStr}&apikey=${apiKey}`
  )
  if (!response.ok) throw new Error(`Price API error: ${response.status}`)
  return response.json()
}

// For single fund sparkline (180 days of history)
export async function fetchFundHistory(apiKey: string, symbol: string, days: number = 180) {
  const response = await fetch(
    `${TWELVE_DATA_BASE}/time_series?symbol=${symbol}&interval=1day&outputsize=${days}&apikey=${apiKey}`
  )
  if (!response.ok) throw new Error(`History API error: ${response.status}`)
  return response.json()
}

// Parse Twelve Data quote response → our fund_prices shape
export function parseTwelveDataQuote(rawQuote: any, fundId: string) {
  return {
    fund_id: fundId,
    price_date: new Date().toISOString().split('T')[0],
    price_native: parseFloat(rawQuote.close || rawQuote.price || 0),
    price_usd: parseFloat(rawQuote.close || rawQuote.price || 0), // handle non-USD separately
    change_1d: parseFloat(rawQuote.change || 0),
    change_1d_percent: parseFloat(rawQuote.percent_change || 0),
    source: 'api' as const
  }
}
```

### src/lib/portfolioCalculations.ts

```typescript
// Calculates portfolio snapshot values from units held × current prices

export function calculatePackageCurrentValue(
  allocations: Array<{
    fund_id: string
    units_held: number
  }>,
  latestPrices: Record<string, number>  // fund_id → price_usd
): number {
  return allocations.reduce((total, alloc) => {
    const price = latestPrices[alloc.fund_id] || 0
    return total + (alloc.units_held * price)
  }, 0)
}

export function calculateTotalInvested(
  transactions: Array<{
    transaction_type: string
    amount_usd: number
  }>
): number {
  return transactions.reduce((total, tx) => {
    const inflows = ['initial_investment','monthly_contribution','top_up','dividend_reinvested']
    const outflows = ['withdrawal','fee_charged']
    if (inflows.includes(tx.transaction_type)) return total + tx.amount_usd
    if (outflows.includes(tx.transaction_type)) return total - tx.amount_usd
    return total  // rebalance transactions don't change invested total
  }, 0)
}

export function buildFundBreakdown(
  allocations: Array<{ fund_id: string; units_held: number; weight_percent: number }>,
  funds: Array<{ id: string; name: string; ticker: string; asset_class: string }>,
  latestPrices: Record<string, number>,
  totalValue: number
) {
  return allocations.map(alloc => {
    const fund = funds.find(f => f.id === alloc.fund_id)
    const priceUsd = latestPrices[alloc.fund_id] || 0
    const valueUsd = alloc.units_held * priceUsd
    return {
      fund_id: alloc.fund_id,
      fund_name: fund?.name || 'Unknown',
      ticker: fund?.ticker || '',
      asset_class: fund?.asset_class || '',
      units: alloc.units_held,
      price_usd: priceUsd,
      value_usd: valueUsd,
      weight_pct: totalValue > 0 ? (valueUsd / totalValue) * 100 : 0
    }
  })
}
```

## SUPABASE EDGE FUNCTION: update-fund-prices

Create: supabase/functions/update-fund-prices/index.ts

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const TWELVE_DATA_KEY = Deno.env.get('TWELVE_DATA_API_KEY')!
const supabase = createClient(
  Deno.env.get('SUPABASE_URL')!,
  Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
)

Deno.serve(async (req) => {
  try {
    // 1. Get all active funds with API symbols
    const { data: funds, error: fundsError } = await supabase
      .from('funds')
      .select('id, name, ticker, price_api_symbol, price_api_exchange, currency')
      .eq('is_active', true)
      .not('price_api_symbol', 'is', null)

    if (fundsError) throw fundsError

    // 2. Batch fetch from Twelve Data (max 100 symbols per request)
    const symbols = funds.map(f => f.price_api_symbol).join(',')
    const apiUrl = `https://api.twelvedata.com/quote?symbol=${symbols}&apikey=${TWELVE_DATA_KEY}`
    const response = await fetch(apiUrl)
    if (!response.ok) throw new Error(`Twelve Data API error: ${response.status}`)
    const priceData = await response.json()

    // 3. Upsert prices
    const today = new Date().toISOString().split('T')[0]
    const upserts = []

    for (const fund of funds) {
      // Twelve Data returns single object if 1 symbol, keyed object if multiple
      const quote = funds.length === 1
        ? priceData
        : priceData[fund.price_api_symbol]

      if (!quote || quote.status === 'error') continue

      const priceNative = parseFloat(quote.close || quote.price || '0')
      const priceUsd = fund.currency === 'USD' ? priceNative : priceNative // extend for GBP/EUR

      upserts.push({
        fund_id: fund.id,
        price_date: today,
        price_native: priceNative,
        price_usd: priceUsd,
        price_vnd: priceUsd * 25400,
        change_1d: parseFloat(quote.change || '0'),
        change_1d_percent: parseFloat(quote.percent_change || '0'),
        source: 'api'
      })
    }

    const { error: upsertError } = await supabase
      .from('fund_prices')
      .upsert(upserts, { onConflict: 'fund_id,price_date' })

    if (upsertError) throw upsertError

    // 4. Recalculate portfolio snapshots for all active packages
    await recalculatePortfolioSnapshots(supabase, today)

    return new Response(
      JSON.stringify({ success: true, updated: upserts.length, date: today }),
      { headers: { 'Content-Type': 'application/json' } }
    )
  } catch (error) {
    return new Response(
      JSON.stringify({ success: false, error: error.message }),
      { status: 500, headers: { 'Content-Type': 'application/json' } }
    )
  }
})

async function recalculatePortfolioSnapshots(supabase: any, today: string) {
  // Fetch all active packages with their allocations
  const { data: packages } = await supabase
    .from('client_packages')
    .select(`
      id, user_id,
      package_allocations (fund_id, units_held),
      package_transactions (transaction_type, amount_usd)
    `)
    .eq('status', 'active')

  // Fetch latest prices
  const { data: latestPrices } = await supabase
    .from('fund_prices')
    .select('fund_id, price_usd')
    .eq('price_date', today)

  const priceMap: Record<string, number> = {}
  for (const p of latestPrices || []) {
    priceMap[p.fund_id] = p.price_usd
  }

  const snapshots = []
  for (const pkg of packages || []) {
    // Calculate current value: sum(units × price)
    const totalValue = (pkg.package_allocations || []).reduce((sum: number, alloc: any) => {
      return sum + (alloc.units_held * (priceMap[alloc.fund_id] || 0))
    }, 0)

    // Calculate total invested from transactions
    const inflowTypes = ['initial_investment','monthly_contribution','top_up','dividend_reinvested']
    const outflowTypes = ['withdrawal','fee_charged']
    const totalInvested = (pkg.package_transactions || []).reduce((sum: number, tx: any) => {
      if (inflowTypes.includes(tx.transaction_type)) return sum + tx.amount_usd
      if (outflowTypes.includes(tx.transaction_type)) return sum - tx.amount_usd
      return sum
    }, 0)

    const returnPct = totalInvested > 0
      ? ((totalValue - totalInvested) / totalInvested) * 100
      : 0

    snapshots.push({
      client_package_id: pkg.id,
      user_id: pkg.user_id,
      snapshot_date: today,
      total_invested_usd: totalInvested,
      total_value_usd: totalValue,
      total_return_percent: returnPct,
      fund_breakdown: JSON.stringify(pkg.package_allocations?.map((alloc: any) => ({
        fund_id: alloc.fund_id,
        units: alloc.units_held,
        price_usd: priceMap[alloc.fund_id] || 0,
        value_usd: alloc.units_held * (priceMap[alloc.fund_id] || 0)
      })))
    })
  }

  if (snapshots.length > 0) {
    await supabase
      .from('portfolio_snapshots')
      .upsert(snapshots, { onConflict: 'client_package_id,snapshot_date' })
  }
}
```

Schedule this Edge Function via Supabase cron:
  Schedule: `0 22 * * 1-5` (22:00 UTC weekdays — after US market close)
  Also expose as HTTP endpoint for manual trigger from admin panel.

## KEY BEHAVIOURS

1. PORTAL STATE GATE: On every portal page load, check client_profiles.status.
   If 'prospect': show ProspectBanner at top of every page + lock portfolio pages.
   If 'active': full access.

2. PRICE FRESHNESS: Show "Prices updated: [date]" on every page showing fund prices.
   If last price_date is >2 business days ago: show amber warning
   "Price data may be delayed — contact your advisor."

3. REAL-TIME MESSAGES: Supabase Realtime subscription in useConversation hook.
   Message notification badge on nav updates without refresh.

4. SNAPSHOT AUTO-CALCULATION: After advisor enters any package_transaction, trigger
   a snapshot recalculation for that package immediately (not waiting for cron).
   Show "Portfolio updated" toast to client.

5. GOAL PROJECTION RECALC: Runs whenever:
   - New portfolio_snapshot is created for a linked package
   - Goal fields (target_amount, target_date) are changed
   - Package expected_annual_return is changed by advisor
   Updates financial_goals.status field automatically.

6. SCENARIO AUTOSAVE: Every scenario calculation automatically saves to scenarios table.
   No "save" button needed — just run it and it's in history.

7. ADVISOR TASK NOTIFICATIONS: When client requests anything (top-up, pause, scenario apply),
   advisor gets in-app notification AND email. Client sees "Pending — advisor notified" chip.

8. ALLOCATION WEIGHT VALIDATION: FundAllocationEditor blocks saving if sum ≠ 100%.
   Shows running total prominently. Prevents bad data entering the system.

9. PROSPECT ILLUSTRATIONS: All projections shown to prospects include the disclaimer:
   "Illustration assumes [X]% annual return based on your risk profile.
    Actual returns vary and are not guaranteed."
   X = Conservative 4.5% / Moderate Conservative 5.5% / Moderate 7% /
       Moderate Aggressive 9% / Aggressive 11%

10. CURRENCY TOGGLE: USD/VND toggle in settings applies globally.
    VND = value_usd × 25,400 (fixed rate). Show "₫" prefix.
    Show FX disclaimer: "VND values use a fixed rate of ₫25,400/USD for illustration."

```typescript
// src/lib/portfolioCalculations.ts

export function calculateTotalInvested(transactions: Array<{
  transaction_type: string
  amount_usd: number
}>): number {
  const INFLOWS = ['initial_investment','monthly_contribution','top_up','dividend_reinvested']
  const OUTFLOWS = ['withdrawal','fee_charged']
  return transactions.reduce((total, tx) => {
    if (INFLOWS.includes(tx.transaction_type)) return total + tx.amount_usd
    if (OUTFLOWS.includes(tx.transaction_type)) return total - tx.amount_usd
    return total  // rebalance transactions don't change invested total
  }, 0)
}

export function calculateCurrentValue(
  allocations: Array<{ fund_id: string; units_held: number }>,
  priceMap: Record<string, number>  // fund_id → price_usd
): number {
  return allocations.reduce((total, alloc) => {
    return total + (alloc.units_held * (priceMap[alloc.fund_id] || 0))
  }, 0)
}

export function buildFundBreakdown(
  allocations: Array<{ fund_id: string; units_held: number; weight_percent: number }>,
  fundDetails: Array<{ id: string; name: string; ticker: string; asset_class: string }>,
  priceMap: Record<string, number>,
  totalValue: number
) {
  return allocations.map(alloc => {
    const fund = fundDetails.find(f => f.id === alloc.fund_id)
    const price = priceMap[alloc.fund_id] || 0
    const value = alloc.units_held * price
    return {
      fund_id: alloc.fund_id,
      fund_name: fund?.name || 'Unknown Fund',
      ticker: fund?.ticker || '',
      asset_class: fund?.asset_class || '',
      units: alloc.units_held,
      price_usd: price,
      value_usd: value,
      weight_pct: totalValue > 0 ? (value / totalValue) * 100 : 0,
      target_weight_pct: alloc.weight_percent
    }
  })
}

// Auto-calculate units from amount + price (for transaction form)
export function unitsFromAmount(amountUsd: number, pricePerUnit: number): number {
  if (pricePerUnit <= 0) return 0
  return parseFloat((amountUsd / pricePerUnit).toFixed(6))
}

// Recalculate and upsert snapshot after a transaction is entered
export async function recalculatePackageSnapshot(
  supabase: any,
  packageId: string,
  userId: string,
  priceMap: Record<string, number>,
  today: string
) {
  // Fetch all transactions for this package
  const { data: transactions } = await supabase
    .from('package_transactions')
    .select('transaction_type, amount_usd')
    .eq('client_package_id', packageId)

  // Fetch allocations with units
  const { data: allocations } = await supabase
    .from('package_allocations')
    .select('fund_id, units_held, weight_percent')
    .eq('client_package_id', packageId)
    .eq('is_active', true)

  const totalInvested = calculateTotalInvested(transactions || [])
  const totalValue = calculateCurrentValue(allocations || [], priceMap)
  const returnPercent = totalInvested > 0
    ? ((totalValue - totalInvested) / totalInvested) * 100
    : 0

  await supabase.from('portfolio_snapshots').upsert({
    client_package_id: packageId,
    user_id: userId,
    snapshot_date: today,
    total_invested_usd: totalInvested,
    total_value_usd: totalValue,
    total_return_percent: returnPercent,
    fund_breakdown: allocations || []
  }, { onConflict: 'client_package_id,snapshot_date' })
}
```

## STEP 8 — SUPABASE EDGE FUNCTION

Create this file: supabase/functions/update-fund-prices/index.ts

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const TWELVE_DATA_KEY = Deno.env.get('TWELVE_DATA_API_KEY')!
const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!
const SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!

const supabase = createClient(SUPABASE_URL, SERVICE_KEY)

Deno.serve(async (req: Request) => {
  // Allow both cron trigger and manual HTTP call (from admin panel)
  const { fundIds } = await req.json().catch(() => ({}))

  try {
    // 1. Fetch funds to update (all active, or specific ids if passed)
    let query = supabase
      .from('funds')
      .select('id, name, ticker, price_api_symbol, currency')
      .eq('is_active', true)
      .not('price_api_symbol', 'is', null)

    if (fundIds && fundIds.length > 0) {
      query = query.in('id', fundIds)
    }

    const { data: funds, error: fundsError } = await query
    if (fundsError) throw fundsError
    if (!funds || funds.length === 0) {
      return Response.json({ success: true, updated: 0, message: 'No funds to update' })
    }

    // 2. Batch call Twelve Data (comma-separated symbols, max 100 per request)
    const symbols = funds.map((f: any) => f.price_api_symbol).join(',')
    const apiResponse = await fetch(
      `https://api.twelvedata.com/quote?symbol=${encodeURIComponent(symbols)}&apikey=${TWELVE_DATA_KEY}`
    )

    if (!apiResponse.ok) {
      throw new Error(`Twelve Data API error: ${apiResponse.status} ${apiResponse.statusText}`)
    }

    const priceData = await apiResponse.json()
    const today = new Date().toISOString().split('T')[0]
    const upserts: any[] = []
    const errors: string[] = []

    for (const fund of funds as any[]) {
      // Single symbol returns object directly; multiple symbols returns keyed object
      const quote = funds.length === 1
        ? priceData
        : priceData[fund.price_api_symbol]

      if (!quote || quote.status === 'error' || !quote.close) {
        errors.push(`${fund.ticker}: ${quote?.message || 'No data returned'}`)
        continue
      }

      const priceNative = parseFloat(quote.close)
      // Twelve Data returns prices in the fund's native currency
      // For simplicity: treat all as USD (extend for GBP/EUR funds if needed)
      const priceUsd = priceNative

      upserts.push({
        fund_id: fund.id,
        price_date: today,
        price_native: priceNative,
        price_usd: priceUsd,
        price_vnd: priceUsd * 25400,
        change_1d: parseFloat(quote.change || '0'),
        change_1d_percent: parseFloat(quote.percent_change || '0'),
        source: 'api'
      })
    }

    if (upserts.length > 0) {
      const { error: upsertError } = await supabase
        .from('fund_prices')
        .upsert(upserts, { onConflict: 'fund_id,price_date' })
      if (upsertError) throw upsertError
    }

    // 3. After prices are updated, recalculate portfolio snapshots for all active packages
    if (!fundIds) {
      // Only do full recalc on the daily cron run (not single-fund manual refresh)
      await recalculateAllActiveSnapshots(today)
    }

    return Response.json({
      success: true,
      updated: upserts.length,
      errors: errors,
      date: today
    })

  } catch (error: any) {
    return Response.json(
      { success: false, error: error.message },
      { status: 500 }
    )
  }
})

async function recalculateAllActiveSnapshots(today: string) {
  const { data: packages } = await supabase
    .from('client_packages')
    .select(`
      id, user_id,
      package_allocations (fund_id, units_held),
      package_transactions (transaction_type, amount_usd)
    `)
    .eq('status', 'active')

  if (!packages || packages.length === 0) return

  // Get latest price for all funds
  const { data: latestPrices } = await supabase
    .from('fund_prices')
    .select('fund_id, price_usd')
    .eq('price_date', today)

  const priceMap: Record<string, number> = {}
  for (const p of latestPrices || []) {
    priceMap[p.fund_id] = p.price_usd
  }

  const INFLOWS = ['initial_investment','monthly_contribution','top_up','dividend_reinvested']
  const OUTFLOWS = ['withdrawal','fee_charged']
  const snapshots = []

  for (const pkg of packages as any[]) {
    const totalInvested = (pkg.package_transactions || []).reduce((sum: number, tx: any) => {
      if (INFLOWS.includes(tx.transaction_type)) return sum + tx.amount_usd
      if (OUTFLOWS.includes(tx.transaction_type)) return sum - tx.amount_usd
      return sum
    }, 0)

    const totalValue = (pkg.package_allocations || []).reduce((sum: number, alloc: any) => {
      return sum + (alloc.units_held * (priceMap[alloc.fund_id] || 0))
    }, 0)

    const returnPct = totalInvested > 0
      ? ((totalValue - totalInvested) / totalInvested) * 100
      : 0

    snapshots.push({
      client_package_id: pkg.id,
      user_id: pkg.user_id,
      snapshot_date: today,
      total_invested_usd: totalInvested,
      total_value_usd: totalValue,
      total_return_percent: returnPct,
      fund_breakdown: pkg.package_allocations || []
    })
  }

  if (snapshots.length > 0) {
    await supabase.from('portfolio_snapshots')
      .upsert(snapshots, { onConflict: 'client_package_id,snapshot_date' })
  }
}
```

## STEP 9 — KEY COMPONENT SPECIFICATIONS

### FundAllocationEditor.tsx
The most critical advisor UI component. Used in TAB 3 of client detail.

STATE:
  allocations: Array<{ fund_id, fund_name, ticker, asset_class, weight_percent, units_held }>
  weightTotal: number (sum of weight_percent — recalculates on every change)
  isDirty: boolean (true if unsaved changes exist)
  isSaving: boolean

UI:
  Table with editable rows. Each row:
    Fund name + ticker (read-only) | Asset class chip |
    Weight % (number input, 0–100, 1 decimal) |
    Units held (number input, 6 decimals) |
    Current price (from useFundPrices, read-only) |
    Value ($) (units × price, calculated live, read-only) |
    [Remove] icon button (trash icon).

  Below table:
    Weight total bar: horizontal progress bar 0–100.
      Fills green when ≈ 100%. Red when over or under.
    Text: "Total: XX.X% / 100%" — red if not ≈ 100%.
    Remaining: "+X.X% remaining" or "−X.X% over allocated".

  [+ Add fund] button: opens FundSearchDropdown inline.
  FundSearchDropdown: text input (debounced search, min 2 chars) → dropdown list.
    Each result: fund name / ticker / asset class chip / last price.
    Click → adds new row to table with weight=0 and units=0.

  [Save allocation] button:
    Disabled (grayed) if weightTotal is not within 99.5–100.5%.
    Loading state while saving.
    On success: "Saved ✓ [timestamp]" message for 3 seconds.

  "Unsaved changes" pill (amber) visible if isDirty=true.

### TransactionForm.tsx
Inline form for entering a transaction. Can be used inside TAB 4 or as a standalone panel.

FIELDS:
  1. Package selector (select from client's active packages).
     On change → loads that package's fund allocations.
  2. Fund selector (select from loaded allocations).
     Shows: fund name + ticker + current price.
     On change → auto-fills price_per_unit from latest fund_prices for today.
  3. Transaction type (styled select).
  4. Date (date picker, default today).
     On date change → refetches fund price for that date (from fund_prices table).
  5. Amount USD (CurrencyInput).
     On change → auto-calculates units: units = amount / price_per_unit.
  6. Price per unit (number input, pre-filled from fund price).
     User can override.
     On change → recalculates units.
  7. Units (number input, 6 decimal places).
     Shows auto-calculated value. User can override (then amount becomes price × units).
  8. Notes (text input, optional).

Auto-calculation rules:
  When amount changes AND price is set: units = amount / price.
  When price changes AND amount is set: recalculate units.
  When units are manually overridden: amount = units × price.

[Save transaction] button → calls useClientPackages.enterTransaction() → shows success/error toast.

### ClientSummaryCard.tsx
Compact card for lists (used in advisor dashboard and admin lists).

Shows: avatar circle (initials, teal bg) + name + email + status chip +
       portfolio value + packages count + risk profile.
Clickable → navigates to /advisor/clients/:id.

### DataTable.tsx
Reusable sortable table. Props: columns, data, onRowClick, loading, emptyState.
Column config: { key, label, sortable, render (optional custom cell renderer) }
Sorted client-side by default. Loading state: skeleton rows.
Mobile: horizontal scroll wrapper.

## KEY BEHAVIOURS TO ENFORCE

1. WEIGHT VALIDATION: FundAllocationEditor [Save] button must remain DISABLED
   until weights sum to exactly 100% (±0.5 tolerance). No exceptions. Show clear
   red error if user tries to click while disabled.

2. SNAPSHOT ON EVERY TRANSACTION: After every package_transactions INSERT or DELETE,
   immediately call recalculatePackageSnapshot(). Client-facing portfolio value must
   always reflect the latest transactions. Never stale.

3. UNITS AUTO-UPDATE: After entering a transaction, the package_allocations.units_held
   must be updated immediately. Buy transactions ADD units. Sell/withdrawal SUBTRACTS.
   Fee transactions do NOT change units.

4. ADMIN SEES ALL: Every admin page bypasses advisor_id filtering. The RLS policies
   enforce this — super_admin sees all client_profiles regardless of advisor_id.
   Verify this in Supabase: a super_admin policy should be present on every table.

5. ADVISOR SEES ONLY ASSIGNED: Advisor pages filter by advisor_id = current user.
   An advisor should NEVER see another advisor's clients or packages.

6. PRICE AUTO-FILL: In TransactionForm, when advisor selects a fund + date, immediately
   query fund_prices for that fund_id and price_date. If found, auto-fill price_per_unit.
   If not found (e.g. weekend/holiday): show "No price on record for this date —
   enter manually" hint.

7. TOAST NOTIFICATIONS: Every save action shows a toast:
   success: green "Saved ✓" | error: red "Failed — [reason]" | 3 second auto-dismiss.

8. LOADING SKELETONS: Every data-fetching section shows LoadingSkeleton while loading.
   Never show blank white areas.

9. EMPTY STATES: Every list/table has a designed empty state with icon + message + CTA.

10. MOBILE RESPONSIVE: SidebarNav collapses to icon-only on tablet, hidden on mobile
    (bottom tabs on mobile — 4 items for advisor: Dashboard/Clients/Tasks/+).
    DataTable wraps in horizontal scroll on mobile.

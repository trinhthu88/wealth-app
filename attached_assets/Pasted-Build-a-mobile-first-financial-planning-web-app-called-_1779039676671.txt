Build a mobile-first financial planning web app called WEALTHAPP for free users. Use React + TypeScript + Vite + Tailwind CSS frontend, with Supabase for auth and database. Deploy-ready on Replit.

---

## TECH STACK

- Frontend: React 18 + TypeScript + Vite
- Styling: Tailwind CSS v3
- Routing: React Router v6
- Database + Auth: Supabase (use environment variables VITE_SUPABASE_URL and VITE_SUPABASE_ANON_KEY)
- Charts: Recharts
- Animations: Framer Motion
- Icons: Lucide React
- HTTP: Supabase JS client only (no axios needed)

Install all dependencies in package.json. Configure Tailwind with the custom colors below.

## DATABASE

Create src/lib/schema.sql with this SQL. Tell the user to paste it into their Supabase SQL editor:

```sql
create table profiles (
  id uuid primary key references auth.users(id) on delete cascade,
  email text not null,
  full_name text,
  role text not null default 'free_user',
  age int,
  life_stage text,
  country_code text,
  is_expat boolean default false,
  preferred_currency text default 'USD',
  risk_profile text,
  pathway_step int default 0,
  pathway_complete boolean default false,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create table pathway_steps (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  step_number int not null,
  step_name text not null,
  status text default 'not_started',
  form_data jsonb default '{}',
  completed_at timestamptz,
  updated_at timestamptz default now(),
  unique(user_id, step_number)
);

create table budget_entries (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  period_month date not null,
  currency text default 'USD',
  income numeric default 0,
  housing numeric default 0,
  food numeric default 0,
  transport numeric default 0,
  utilities numeric default 0,
  entertainment numeric default 0,
  other_expenses numeric default 0,
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create table financial_goals (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  title text not null,
  goal_type text not null,
  target_amount numeric default 0,
  current_amount numeric default 0,
  monthly_contribution numeric default 0,
  target_date date,
  currency text default 'USD',
  status text default 'on_track',
  created_at timestamptz default now(),
  updated_at timestamptz default now()
);

create table net_worth_snapshots (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  snapshot_date date default current_date,
  total_savings numeric default 0,
  total_investments numeric default 0,
  total_debts numeric default 0,
  created_at timestamptz default now()
);

create table health_scores (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  score_date date default current_date,
  overall_score int default 0,
  budget_score int default 0,
  goals_score int default 0,
  savings_score int default 0,
  plan_score int default 0,
  insight_1 text,
  insight_2 text,
  insight_3 text,
  created_at timestamptz default now()
);

create table habit_checkins (
  id uuid primary key default gen_random_uuid(),
  user_id uuid references profiles(id) on delete cascade,
  checkin_date date default current_date,
  current_streak int default 0,
  created_at timestamptz default now()
);

alter table profiles enable row level security;
create policy "own_profile" on profiles for all to authenticated using (id = auth.uid());

alter table pathway_steps enable row level security;
create policy "own_steps" on pathway_steps for all to authenticated using (user_id = auth.uid());

alter table budget_entries enable row level security;
create policy "own_budget" on budget_entries for all to authenticated using (user_id = auth.uid());

alter table financial_goals enable row level security;
create policy "own_goals" on financial_goals for all to authenticated using (user_id = auth.uid());

alter table net_worth_snapshots enable row level security;
create policy "own_nw" on net_worth_snapshots for all to authenticated using (user_id = auth.uid());

alter table health_scores enable row level security;
create policy "own_scores" on health_scores for all to authenticated using (user_id = auth.uid());

alter table habit_checkins enable row level security;
create policy "own_checkins" on habit_checkins for all to authenticated using (user_id = auth.uid());
```

## HOOKS

### src/hooks/useAuth.ts
```ts
import { useEffect, useState } from 'react'
import { supabase } from '../lib/supabase'
import { Session } from '@supabase/supabase-js'

export function useAuth() {
  const [session, setSession] = useState<Session | null>(null)
  const [loading, setLoading] = useState(true)

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setSession(data.session)
      setLoading(false)
    })
    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session)
    })
    return () => subscription.unsubscribe()
  }, [])

  return { session, loading }
}
```

### src/hooks/useProfile.ts
Fetch and update the current user's profile from the profiles table.
Exposes: profile, loading, updateProfile(updates).
Auto-fetches when session changes.

### src/hooks/useBudget.ts
Fetch current month budget_entries row for the current user.
Exposes: budget, loading, saveBudget(data), currentMonth.
Calculates savingsRate = ((income - total_expenses) / income) * 100.

### src/hooks/useGoals.ts
Fetch all financial_goals for current user.
Exposes: goals, loading, createGoal(data), updateGoal(id, data), addContribution(goalId, amount).
addContribution increments current_amount by amount and calls updateGoal.

### src/hooks/useHealthScore.ts
Fetch latest health_scores row for current user.
Exposes: score, loading, calculateAndSave().
calculateAndSave computes overall_score from:
  - plan_score: (pathway_step / 6) * 25 — capped at 25
  - savings_score: savingsRate > 20 = 25, 10-20 = 15, < 10 = 5
  - goals_score: goals[0] exists with target_amount > 0 = 25, exists = 15, none = 0
  - budget_score: budget exists with income > 0 = 25, else 0
  overall_score = sum of all four
Then inserts new row to health_scores.

### src/hooks/usePathway.ts
Fetch all pathway_steps for current user.
Exposes: steps, currentStep, loading, saveStep(stepNumber, formData, status), completionPercent.
completionPercent = (completed steps count / 6) * 100.
saveStep does an upsert on (user_id, step_number).

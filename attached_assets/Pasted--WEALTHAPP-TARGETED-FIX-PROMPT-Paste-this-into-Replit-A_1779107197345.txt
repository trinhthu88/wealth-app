## FIX 3 — src/hooks/useGoals.ts
# recalculateProjections() must use rates from profiles, not hardcoded values.
# It must also receive monthlyCashSaved from the current month's budget.

The recalculateProjections function must accept current budget data and use profile rates:

```typescript
async function recalculateProjections(
  goals: Goal[],
  profile: Profile,
  monthlyCashSaved: number
) {
  for (const goal of goals) {
    if (!goal.target_date || !goal.target_amount) continue

    const result = calculateProjection({
      currentAmount: goal.current_amount ?? 0,
      targetAmount: goal.target_amount,
      targetDate: goal.target_date,
      monthlyCashSaved: monthlyCashSaved,                          // FROM BUDGET, NOT ZERO
      savingsBalance: profile.total_savings ?? 0,                  // FROM PROFILES
      savingsRatePercent: profile.savings_rate_percent ?? 4.0,     // FROM PROFILES
      investmentValue: profile.total_investments ?? 0,             // FROM PROFILES
      investmentRatePercent: profile.investment_rate_percent ?? 7.0 // FROM PROFILES
    })

    await supabase.from('financial_goals').update({
      status: result.status,
      on_track: result.onTrack,
      required_monthly_saving: result.requiredMonthlySaving,
      projected_amount_at_target: result.projectedAmount,
      monthly_shortfall: result.monthlyShortfall,
      months_remaining: result.monthsRemaining,
      gap_percent: result.gapPercent,
      last_projection_update: new Date().toISOString(),
    }).eq('id', goal.id)
  }
  // Re-fetch goals after update
  fetchGoals()
}
```

TRIGGER: Call recalculateProjections() whenever:
- monthlyCashSaved changes (budget updated)
- profile.savings_rate_percent changes (settings updated)
- profile.investment_rate_percent changes (settings updated)
- profile.total_savings changes (settings updated)
- profile.total_investments changes (settings updated)

```typescript
const recalculateSummary = async () => {
  const income = transactions
    .filter(t => t.category === 'income')
    .reduce((sum, t) => sum + Math.abs(t.amount), 0)

  const expenses = transactions
    .filter(t => t.category !== 'income')
    .reduce((sum, t) => sum + Math.abs(t.amount), 0)

  const netSavings = income - expenses
  const savingsRate = income > 0 ? (netSavings / income) * 100 : 0

  await supabase.from('budget_monthly_summaries').upsert({
    user_id: userId,
    period_month: selectedMonth,   // format: 'YYYY-MM-01'
    total_income: income,
    total_expenses: expenses,
    net_savings: netSavings,
    savings_rate: savingsRate,
    updated_at: new Date().toISOString(),
  }, { onConflict: 'user_id,period_month' })
}
```

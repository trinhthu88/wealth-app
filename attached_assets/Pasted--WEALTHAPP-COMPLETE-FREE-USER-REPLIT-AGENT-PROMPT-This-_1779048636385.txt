### src/lib/goalProjection.ts

```typescript
export interface ProjectionInput {
  currentAmount: number
  targetAmount: number
  targetDate: string
  monthlyCashSaved: number
  savingsBalance: number
  savingsRatePercent: number
  investmentValue: number
  investmentRatePercent: number
}

export interface ProjectionResult {
  status: 'on_track' | 'almost' | 'off_track'
  onTrack: boolean
  monthsRemaining: number
  requiredMonthlySaving: number
  projectedAmount: number
  monthlyShortfall: number
  amountStillNeeded: number
  projectedShortfall: number
  gapPercent: number
  monthlyGrowthFromReturns: number
}

export function calculateProjection(input: ProjectionInput): ProjectionResult {
  const {
    currentAmount, targetAmount, targetDate,
    monthlyCashSaved, savingsBalance, savingsRatePercent,
    investmentValue, investmentRatePercent
  } = input

  const today = new Date()
  const target = new Date(targetDate)
  const monthsRemaining = Math.max(0,
    (target.getFullYear() - today.getFullYear()) * 12 +
    (target.getMonth() - today.getMonth())
  )

  const amountStillNeeded = Math.max(0, targetAmount - currentAmount)
  const requiredMonthlySaving = monthsRemaining > 0
    ? amountStillNeeded / monthsRemaining : amountStillNeeded

  const savingsMonthlyRate = savingsRatePercent / 100 / 12
  const investMonthlyRate = investmentRatePercent / 100 / 12

  const projectedInterest = savingsBalance *
    (Math.pow(1 + savingsMonthlyRate, monthsRemaining) - 1)
  const projectedInvestmentGrowth = investmentValue *
    (Math.pow(1 + investMonthlyRate, monthsRemaining) - 1)
  const projectedCashSavings = monthlyCashSaved * monthsRemaining

  const projectedAmount = currentAmount + projectedCashSavings +
    projectedInterest + projectedInvestmentGrowth

  const projectedShortfall = Math.max(0, targetAmount - projectedAmount)
  const gapPercent = targetAmount > 0 ? (projectedShortfall / targetAmount) * 100 : 0
  const monthlyShortfall = Math.max(0, requiredMonthlySaving - monthlyCashSaved)
  const monthlyGrowthFromReturns =
    savingsBalance * savingsMonthlyRate + investmentValue * investMonthlyRate

  const onTrack = projectedAmount >= targetAmount
  const status = onTrack ? 'on_track' : gapPercent <= 20 ? 'almost' : 'off_track'

  return {
    status, onTrack, monthsRemaining, requiredMonthlySaving,
    projectedAmount, monthlyShortfall, amountStillNeeded,
    projectedShortfall, gapPercent, monthlyGrowthFromReturns
  }
}

export function applyMonthlyGrowth(
  currentAmount: number,
  monthlyCashSaved: number,
  savingsBalance: number,
  savingsRatePercent: number,
  investmentValue: number,
  investmentRatePercent: number
): number {
  const monthlyInterest = savingsBalance * (savingsRatePercent / 100 / 12)
  const monthlyInvestGain = investmentValue * (investmentRatePercent / 100 / 12)
  return currentAmount + monthlyCashSaved + monthlyInterest + monthlyInvestGain
}
```

### src/lib/healthScore.ts

```typescript
export interface HealthScoreInput {
  income: number
  totalExpenses: number
  savingsBalance: number
  savingsRatePercent: number
  investmentValue: number
  investmentRatePercent: number
  netWorth: number
  prevMonthNetWorth: number
  goal: any | null
  goalProjection: any | null
  pathwayStepsCompleted: number
}

export interface HealthScoreResult {
  overall: number
  savingsScore: number
  goalsScore: number
  netWorthScore: number
  wealthGrowthScore: number
  planScore: number
  effectiveSavingsRate: number
  insight1: string
  insight2: string
  insight3: string
}

export function calculateHealthScore(input: HealthScoreInput): HealthScoreResult {
  const {
    income, totalExpenses, savingsBalance, savingsRatePercent,
    investmentValue, investmentRatePercent, netWorth, prevMonthNetWorth,
    goal, goalProjection, pathwayStepsCompleted
  } = input

  const monthlyCashSaved = income - totalExpenses
  const monthlyInterest = savingsBalance * (savingsRatePercent / 100 / 12)
  const monthlyInvestGain = investmentValue * (investmentRatePercent / 100 / 12)
  const effectiveMonthlyGain = monthlyCashSaved + monthlyInterest + monthlyInvestGain
  const effectiveSavingsRate = income > 0 ? (effectiveMonthlyGain / income) * 100 : 0

  // Component 1: Effective savings rate (25 pts)
  const savingsScore =
    effectiveSavingsRate >= 25 ? 25 :
    effectiveSavingsRate >= 20 ? 22 :
    effectiveSavingsRate >= 15 ? 18 :
    effectiveSavingsRate >= 10 ? 12 :
    effectiveSavingsRate >= 5  ?  6 : 0

  // Component 2: Goal on-track status (25 pts)
  let goalsScore = 0
  if (goalProjection) {
    goalsScore = goalProjection.status === 'on_track' ? 25
      : goalProjection.status === 'almost' ? 15 : 5
  } else if (goal) {
    goalsScore = 10
  }

  // Component 3: Net worth trend (20 pts)
  const netWorthChange = netWorth - prevMonthNetWorth
  const netWorthScore =
    netWorthChange > 1000 ? 20 :
    netWorthChange > 0    ? 15 :
    netWorthChange === 0  ?  8 : 0

  // Component 4: Wealth growth rate (20 pts)
  const annualWealthGrowth =
    monthlyCashSaved * 12 +
    savingsBalance * (savingsRatePercent / 100) +
    investmentValue * (investmentRatePercent / 100)
  const wealthGrowthRate = netWorth > 0 ? (annualWealthGrowth / netWorth) * 100 : 0
  const wealthGrowthScore =
    wealthGrowthRate >= 10 ? 20 :
    wealthGrowthRate >= 7  ? 16 :
    wealthGrowthRate >= 4  ? 10 :
    wealthGrowthRate >= 1  ?  5 : 0

  // Component 5: Plan completion (10 pts)
  const planScore = Math.round((pathwayStepsCompleted / 6) * 10)

  const overall = Math.min(100,
    savingsScore + goalsScore + netWorthScore + wealthGrowthScore + planScore)

  // Generate insights from real data
  const insights: string[] = []
  if (effectiveSavingsRate < 15) {
    insights.push(`Your effective savings rate is ${effectiveSavingsRate.toFixed(0)}% — including returns. Reaching 20% would significantly accelerate your financial goals.`)
  } else {
    insights.push(`You're saving ${effectiveSavingsRate.toFixed(0)}% effectively including investment returns. You're ahead of most people your age.`)
  }

  if (goal && goalProjection) {
    if (goalProjection.status === 'on_track') {
      insights.push(`Your "${goal.title}" goal is on track — you're projected to reach it by your target date.`)
    } else {
      insights.push(`Your "${goal.title}" goal needs attention — you're projected to fall $${Math.round(goalProjection.projectedShortfall).toLocaleString()} short.`)
    }
  } else {
    insights.push('Setting a financial goal with a target date unlocks 25 more points on your health score.')
  }

  if (netWorthChange > 0) {
    insights.push(`Your net worth grew $${Math.round(netWorthChange).toLocaleString()} this month. Keep building.`)
  } else if (netWorth > 0) {
    insights.push(`Your net worth is positive at $${Math.round(netWorth).toLocaleString()}. Focus on growing it further each month.`)
  } else {
    insights.push('A positive net worth is the foundation of financial health. Focus on reducing debts while growing savings.')
  }

  return {
    overall, savingsScore, goalsScore, netWorthScore,
    wealthGrowthScore, planScore, effectiveSavingsRate,
    insight1: insights[0], insight2: insights[1], insight3: insights[2]
  }
}
```

```typescript
  { key: 'saver_20',          label: '20% saver',         emoji: '💰', desc: 'Reach 20% effective savings rate',        check: (d: any) => d.effectiveSavingsRate >= 20 },

  { key: 'score_70',          label: 'Score 70+',         emoji: '⭐', desc: 'Health score above 70',                   check: (d: any) => d.healthScore >= 70 },
  { key: 'score_80',          label: 'Score 80+',         emoji: '🌟', desc: 'Health score above 80',                   check: (d: any) => d.healthScore >= 80 },
  { key: 'budget_3months',    label: 'Budget habit',      emoji: '📊', desc: 'Updated budget 3 months in a row',        check: (d: any) => d.budgetMonthsCount >= 3 },
]

export async function checkAndAwardMilestones(
  supabase: any, userId: string, data: any, earnedKeys: string[]
): Promise<string[]> {
  const newlyEarned: string[] = []
  for (const m of MILESTONES) {
    if (earnedKeys.includes(m.key)) continue
    if (m.check(data)) {
      const { error } = await supabase.from('user_milestones')
        .upsert({ user_id: userId, milestone_key: m.key },
                 { onConflict: 'user_id,milestone_key' })
      if (!error) newlyEarned.push(m.key)
    }
  }
  return newlyEarned
}
```

```typescript
export function generateWeeklyTip(data: {
  effectiveSavingsRate: number
  healthScore: number
  goal: any | null
  goalStatus: string
  ageBenchmarkSavingsRate: number
  streakWeeks: number
  isExpat: boolean
  investmentRatePercent: number
}) {
  const { effectiveSavingsRate, healthScore, goal, goalStatus,
          ageBenchmarkSavingsRate, streakWeeks, isExpat, investmentRatePercent } = data

  if (goal && goalStatus === 'off_track') return {
    type: 'goal_alert',
    title: 'Your goal needs attention',
    body: `Your "${goal.title}" is off track. An advisor can help close the gap without requiring you to save more each month.`,
    actionLabel: 'See breakdown',
    actionRoute: '/app/goals'
  }

  if (goal && goalStatus === 'almost') return {
    type: 'goal_alert',
    title: "You're almost on track",
    body: `Your "${goal.title}" is within reach — just $${Math.round(goal.monthly_shortfall).toLocaleString()}/month more would put you fully on track.`,
    actionLabel: 'Update my budget',
    actionRoute: '/app/budget'
  }

  if (effectiveSavingsRate < ageBenchmarkSavingsRate) return {
    type: 'benchmark',
    title: 'Your savings vs your peers',
    body: `You're saving ${effectiveSavingsRate.toFixed(0)}% effectively. People your age average ${ageBenchmarkSavingsRate}%. Closing this gap adds years to your retirement date.`,
    actionLabel: 'Review my budget',
    actionRoute: '/app/budget'
  }

  if (isExpat && investmentRatePercent < 5) return {
    type: 'expat_tip',
    title: 'Are your investments expat-optimised?',
    body: 'As an expat, local investment accounts may offer better returns than home country accounts. An advisor can review your options.',
    actionLabel: 'Book a free review',
    actionRoute: '/book'
  }

  if (healthScore < 60) return {
    type: 'score_tip',
    title: 'Three quick wins for your score',
    body: `Your score is ${healthScore}/100. Reach 20% savings (+5 pts), complete your pathway (+5 pts), and set a goal target date (+5 pts).`,
    actionLabel: 'See my score',
    actionRoute: '/app/health-score'

  if (streakWeeks >= 4) return {
    type: 'milestone',
    title: `${streakWeeks}-week streak — you're building something real`,
    body: 'Consistent financial tracking is one of the strongest predictors of long-term wealth. You\'re in the top 20% for engagement.',
    actionLabel: 'View achievements',
    actionRoute: '/app/dashboard'
  }

  return {
    type: 'market_context',
    title: 'Are your savings working hard enough?',
    body: 'Vietnam bank deposit rates average 4.5–5.2% in 2025. Make sure your savings rate reflects what you\'re actually earning.',
    actionLabel: 'Update savings rate',
    actionRoute: '/app/settings'
  }
}
```

```typescript
// Fetches budget_transactions for current month (default) or selected month.
// Exposes:
//   transactions: all rows for month
//   recurring: transactions where is_recurring = true
//   oneOff: transactions where is_recurring = false
//   pendingConfirm: transactions where recurring_status = 'pending_confirm'
//   totalIncome: sum of positive amounts
//   totalExpenses: sum of negative amounts (absolute value)
//   monthlyCashSaved: totalIncome - totalExpenses
//   savingsRate: (monthlyCashSaved / totalIncome) * 100
//   loading, selectedMonth, setSelectedMonth
//   addTransaction(data): inserts new row
//   updateTransaction(id, data): updates row
//   deleteTransaction(id): deletes row
//   confirmAllRecurring(): updates all pending_confirm rows to confirmed for this month
//   recalculateSummary(): upserts budget_monthly_summaries for current month

// When month changes to current month and pending_confirm rows exist,
// the pendingConfirm array will be non-empty, triggering RecurringConfirmBanner.

// Auto-copy recurring logic runs client-side on first load of a new month:
// 1. Check if any budget_monthly_summaries row exists for this month
// 2. If not, find all confirmed recurring transactions from last month
// 3. Insert them for this month with recurring_status = 'pending_confirm'
// 4. Create budget_monthly_summaries row for this month
```

```typescript
// Fetches health_scores for current user (latest row).
// Exposes: score, history (last 6 rows), loading, calculateAndSave()

// calculateAndSave():
//   Gathers: income+expenses from useBudget, savings/investments from profile,
//            net worth from latest net_worth_snapshots,
//            prev month net worth from second-latest snapshot,
//            goal + goalProjection from useGoals,
```

## KEY BEHAVIOURS

1. AUTO-SAVE: pathway fields debounce 800ms → supabase upsert → "Saving... / Saved ✓".
2. REAL-TIME PROJECTION: recalculateProjections() runs whenever monthlyCashSaved,
   savings_rate_percent, or investment_rate_percent changes in profile.
3. MONTHLY GROWTH: on first load of a new month, applyMonthlyGrowth() runs on each goal
   before recalculating projections. Detected by comparing last_projection_update month.
4. RECURRING COPY: on first load of a new month budget page, copies recurring transactions
   from last month with pending_confirm status.
5. CURRENCY: all amounts stored in USD. VND display = amount × 25000. Toggle updates all displays.
6. EMPTY STATES: every section has a designed empty state. No blank white sections ever.
7. LOADING STATES: LoadingSkeleton shown for every section while data fetches.
8. HEALTH SCORE RECALC: triggered on pathway completion, monthly first visit,
   and when user updates financial rates in settings.
9. MILESTONE CHECK: runs after every dashboard load. Awards new milestones automatically.
10. WEEKLY TIP: generateWeeklyTip() runs on dashboard load. Cached in user_tips table
    (one per user per week). Same tip shows all week.
